# Debug menu hooks. See docs/modding.md for the format.

# End of FieldUpdateFrame: HudRefreshStats() -> DebugFieldTick(), which calls it first.
call 0x080470FC sub_0802D8DC DebugFieldTick

# Overworld walking speed: FieldUpdateFrame's bl sub_08041E88 (dir, step).
call 0x080470C2 sub_08041E88 DebugMoveStep

# Part wear after a battle (infinite ripcord / launcher): both callers.
call 0x0803B826 sub_0803D51C DebugPartsApplyWear
call 0x0803BA32 sub_0803D51C DebugPartsApplyWear

# Launch RPM formula (max RPM).
call 0x0803C230 _0803CECC DebugLaunch

# Do not lose the equipped beyblade after a loss (infinite beyblade health).
0x080380E0 KeepBladeOnLoss__Hook

# Experience brackets: level 16 and a full bar at 4000+ experience.
sub_08042BE8 ExpBracketFixed
sub_0802E1B4 ExpBarFillFixed

# The menu state needs EWRAM that the retail heap does not hand out.
# HeapAlloc (0x0806A3DE) and its stats print (0x0806A622) size the heap as
# 0xFE << 10 bytes from 0x02000800, i.e. up to the end of EWRAM. A play-through
# peaks at ~67 KB (tools/ram_survey.py), so make it 0x80 << 10 = 128 KB; that
# leaves 0x02020800..0x02040000 for mods (see asm/ram_map_ewram.s).
patch 0x0806A3DE 80 22 expect FE 22
patch 0x0806A622 80 22 expect FE 22

# All Locations: sub_08043980 recolours the map art unless the path's event
# flag is set; this bl tests those flags.
call 0x080439B6 sub_08045D3C DebugMapFlag

# Character: MapLoadObjects loads the player palette (slot 0, two branches) on
# every map.
call 0x080444E0 sub_08062A74 DebugObjPalLoad
call 0x080444F8 sub_08062A74 DebugObjPalLoad

# Character: script opcode 29 draws a dialogue portrait from a template and
# palette in the script; the wrapper swaps in the chosen person for the
# player's portrait. Entry 29 of the opcode table at 0x08099710.
pointer 0x08099784 sub_0805A304 DebugPortraitOp

# Save: the entries' state is kept in EEPROM blocks 0x3EF.. (after the slot).
# sub_08044DAC (save commit) writes the slot's data blocks through
# sub_08044F14, then the record; sub_08045128 loads the slot, then the record.
call 0x08044E04 sub_08044F14 DebugSaveBlocks
call 0x0804515C sub_08045590 DebugSaveRead

# Abilities (abilities.c). All of them wrap a retail call and work on the human
# player only.
# The battle frame (sub_08031D30) runs each fighter's input handler: Steel Wall
# halves the spin that attacking and dodging cost inside it, and Turbine counts
# its charge here.
call 0x08031D82 _08034BE0 AbilityHandler
call 0x08031D90 _08034BE0 AbilityHandler
# The handler steers the blade by the d-pad (Turbine's speed).
call 0x08034C10 _08030638 TurbineSteer
call 0x08034D08 _08030638 TurbineSteer
# Clash resolution: Siphon takes its share of the damage, Turbine adds to the
# power of an attack and is lost.
call 0x08032B24 _0802FFAC AbilityClash
call 0x0802FFC0 _080300D4 TurbinePower
call 0x0802FFC8 _080300D4 TurbinePower
# Blades in contact fill the bit beast gauge (Gunner).
call 0x08031F90 _08032A88 AbilityContact
# The per-fighter frame update counts the post-attack recovery down (Rocket).
call 0x0803021A sub_080348E8 AbilityUpdate
call 0x08030224 sub_080348E8 AbilityUpdate

# Weather (weather.c). Rain, snow and wind change how the bodies move: the motion
# step of sub_080348E8 (BeybladeUpdate) runs for both blades. The d-pad push of
# rain goes through TurbineSteer, heat and the particles through the ability
# wraps above.
call 0x08034926 sub_08035984 WeatherMotion
