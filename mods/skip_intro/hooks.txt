# Boot sequence (sub_080663C8): after clearing state it calls 0x08065C7C, which
# shows the six splash screens in order (Nintendo licence, copyright, Nelvana,
# D Rights, Atari, Fullfat; table at 0x080B9000) through 0x08065CD0, then goes
# on to the title screen. The first screen also runs SaveDataVerify (0x08044A8C,
# the only place it is called), so the replacement does just that.
call 0x080663E0 0x08065C7C SkipIntro
